% --- Slide 3: Representativeness ---
\begin{frame}{Psychology of Beliefs: Representativeness}
\begin{itemize}
\item Consider questions such as:
  \begin{itemize}
  \item What is the probability that object A comes from class B?
  \item What is the probability that event A was generated from process B?
  \end{itemize}
\item Kahneman and Tversky (1974) argue that people often answer with the ``representativeness heuristic''
  \begin{itemize}
  \item Evaluate the probability by the extent to which A is representative of B
  \item i.e.\ the degree to which A reflects the essential characteristics of B
  \end{itemize}
\item This is helpful in general but can lead to serious biases
\end{itemize}
\end{frame}

% --- Slide 4: Base-rate neglect ---
\begin{frame}{Representativeness, ctd.}
\begin{itemize}
\item \textbf{Base-rate neglect}
\item Consider the following description of Steve. Do you think Steve is more likely to be a librarian or a lawyer?
  \begin{itemize}
  \item Steve is very shy and withdrawn, invariably helpful, but with little interest in people or in the world of reality.
  \item A meek and tidy soul, he has a need for order and structure, and a passion for detail.
  \end{itemize}
\item People judge Steve to be more likely a librarian than a lawyer
\item But there are many more lawyers than librarians in the U.S.
\item The question actually asks about the LHS, but people answer using only the likelihood ratio (the first term on the RHS); this is the root cause of base-rate neglect.
\[
\frac{p(\text{library}\mid\text{data})}{p(\text{lawyer}\mid\text{data})}
=
\frac{p(\text{data}\mid\text{library})}{p(\text{data}\mid\text{lawyer})}
\frac{p(\text{library})}{p(\text{lawyer})}.
\] 
\end{itemize}
\end{frame}

% --- Slide 5: Sample size neglect ---
\begin{frame}{Representativeness, ctd.}
\begin{itemize}
\item \textbf{Sample size neglect}
\item When we do not know the model generating the data, we may infer it too quickly based on too little data
\item This can lead to over-extrapolation of recent trends
\item E.g., the hot hand fallacy
  \begin{itemize}
  \item Seeing a basketball player make a few shots in a row leads to the belief that he is ``hot''
  \item But empirical evidence suggests that the hot hand is largely a myth (Gilovich, Vallone, and Tversky, 1985; but see Miller and Sanjurjo, 2018)
  \end{itemize}
\end{itemize}
\end{frame}


% --- Slide 6: Overconfidence ---
\begin{frame}{Psychology of Beliefs: Overconfidence}
\begin{itemize}
\item There are at least two distinct types of overconfidence:
\item \textbf{Overprecision}
  \begin{itemize}
  \item People are too confident in the accuracy of their forecasts
  \item 98\% confidence intervals include the true quantity around 60\% of the time (Alpert and Raiffa, 1982)
  \end{itemize}
\item \textbf{Overplacement}
  \begin{itemize}
  \item People think they are better than others
  \item 90\% of drivers think they are above-average drivers (Svenson, 1981)
  \end{itemize}
\end{itemize}
\end{frame}

% --- Slide 10: Momentum and Overconfidence ---
\begin{frame}{Overconfidence: Momentum}
\begin{itemize}
\item Daniel, Hirshleifer, and Subrahmanyam (1998)
\item Momentum is driven by investor overconfidence and self-attribution bias
  \begin{itemize}
  \item Investors are overconfident in the private information they gather
  \item Investors also suffer from self-attribution bias: they credit themselves for successes but blame external factors for failures
  \end{itemize}
\item Positive returns reinforce overconfidence, leading to more aggressive trading and price momentum
\item Eventually, prices correct when information is revealed
\end{itemize}
\end{frame}

% --- Slide 12: Over-extrapolation ---
\begin{frame}{Psychology of Beliefs: Over-extrapolation}
\begin{itemize}
\item The estimate of the future value of a quantity is an increasing function of recent past values of that quantity
\item One of the most important and widely applied ideas in behavioral finance
\item The source of over-extrapolation: the representativeness heuristic
\item Application to finance:
  \begin{itemize}
  \item Return extrapolation: predict future returns based on past returns
  \item Fundamentals extrapolation: predict future earnings/cash flows based on past growth
  \end{itemize}
\end{itemize}
\end{frame}

% --- Slide 12a: A simple model (from Lecture 7, pp. 13-16) ---
\begin{frame}[shrink=5]{Over-extrapolation: A Simple Model}
\begin{itemize}
\item \textbf{An economy with $T+1$ periods, $t = 0, 1, \ldots, T$, and two assets}
  \begin{itemize}
  \item A riskless asset with a constant return of zero
  \item A risky asset that is a claim to a single final cash flow $\tilde{D}_T = D_0 + \tilde{\varepsilon}_1 + \ldots + \tilde{\varepsilon}_T$, with $\tilde{\varepsilon}_t \sim N(0, \sigma^2_\varepsilon)$ i.i.d.
  \end{itemize}
\item \textbf{Some investors have extrapolative beliefs about price changes:}
\begin{equation*}
\mathbb{E}^e_t(P_{t+1} - P_t) = X_t \equiv (1-\theta) \sum_{k=1}^{t-1} \theta^{k-1} (P_{t-k} - P_{t-k-1}) + \theta^{t-1} X_1
\end{equation*}
\item \textbf{Other investors --- ``fundamental traders'' --- buy when prices are low relative to expected cash flows, and sell when they are high}
\item \textbf{Equilibrium prices (Barberis, 2018):}
\begin{equation*}
P_t = D_t + \frac{\hat{\mu}}{\hat{\mu}_f} X_t - \gamma \sigma^2_\varepsilon Q\left(T - t - 1 + \frac{1}{\hat{\mu}_f}\right)
\end{equation*}
\end{itemize}
\end{frame}

% --- Slide 12d: Open questions on return extrapolation ---
\begin{frame}{Over-extrapolation: Open Questions}
\begin{itemize}
\item \textbf{What is the source of return extrapolation?}
  \begin{itemize}
  \item And can knowledge of the source deepen our understanding of the empirical facts?
  \end{itemize}
\item \textbf{What value of $\theta$ is consistent with observed prices, and is this value justified?}
\item \textbf{Does $\theta$ vary over time, and if so, why?}
  \begin{itemize}
  \item Cassella and Gulen (2018)
  \end{itemize}
\item \textbf{Are return extrapolation models consistent with the facts about investor portfolios?}
  \begin{itemize}
  \item Who are the investors who are actually trading according to extrapolative beliefs?
  \item Giglio, Maggiori, Stroebel, and Utkus (2021) find that investors' holdings are relatively insensitive to their beliefs
  \item So can shifts in beliefs affect holdings enough to move prices?
  \item Is this investor behavior really about beliefs --- or a different, non-belief mechanism, e.g.\ observational learning?
  \end{itemize}
\end{itemize}
\end{frame}

% --- Slide 16: DOX ---
\begin{frame}[shrink=13]{Over-extrapolation of Past Returns: Applications}
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\begin{itemize}
\item Using survey data on expectations of stock returns, Casella and Gulen (RFS 2018) estimate the degree of extrapolative thinking in investors' beliefs (DOX) about market returns
\item DOX ($1-\lambda$) measures the weight investors place on past returns
\end{itemize}
\[
Exp_t = a + b \sum_{i=0}^{\infty} w_i\, R_{t-(i+1)\Delta t,\, t-i\Delta t},
\]

\[
w_i = \frac{\lambda^{i}}{\displaystyle\sum_{k=0}^{\infty} \lambda^{k}}, \quad 0 \le \lambda < 1,
\]
\begin{itemize}
\item A lower $\lambda$ implies that investors place higher weight on more recent observations
\item Earlier observations contribute less to an extrapolator's expectations
\end{itemize}
\end{column}
\begin{column}{0.40\textwidth}
\centering
\includegraphics[width=\textwidth]{figures/slide16_pic4.png}
\end{column}
\end{columns}
\end{frame}

% --- Slide 17: DOX and Predictability ---
\begin{frame}{Over-extrapolation of Past Returns: Applications, ctd.}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{itemize}
\item They find the predictability of stock returns by a price-scaled variable (e.g., D/P, B/M, or E/P) is contingent on the DOX
\item When DOX is high, investors more quickly ``forget'' the positive price changes that drove up valuations, and their beliefs mean-revert faster
\item As a result, high valuations are more strongly associated with low future returns when DOX is high
\end{itemize}
\end{column}

\begin{frame}[shrink=22]{Extrapolative Beliefs in the Cross-Section: Setup and Data}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{Question.} How do investors form expectations about \emph{individual stock} returns --- and are those expectations rational?

\vspace{0.3em}
\textbf{Data: Forcerank crowdsourcing platform} (Feb 2016--Dec 2017):
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item Rank 10 same-industry stocks by expected return over the \emph{next week}; blind setting rules out herding.
  \item 1,396 contests, 293 tickers, 12,798 rankings, 1,045 users (72 financial professionals).
  \item Consensus score $=$ average ranking $\Rightarrow$ investor expectation proxy.
\end{itemize}

\vspace{0.3em}
\textbf{Key object.} Exponential-decay beliefs:
\[
\mathbb{E}^e_i[R_{i,t+1}] = \lambda_1 \sum_{s=0}^{11} w_s R_{i,t-s},
\qquad w_s = \frac{\lambda_2^s}{\sum_j \lambda_2^j},\quad 0\le\lambda_2<1
\]
$\lambda_1$: level of extrapolation; $\lambda_2$: recency (lower $\Rightarrow$ more weight on recent returns).

\begin{frame}[shrink=16]{Extrapolative Beliefs: Asymmetry and Heterogeneity}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{Negative returns weigh more and decay slower.}
\begin{itemize}
  \item Overall: return 4 weeks back $\approx$ 9\% as important as last week's.
  \item After \emph{negative} returns: 4 weeks back $\approx$ 45\% as important --- investors dwell on losses.
  \item Same asymmetry at the market level: $\lambda_2$ higher in down weeks (cf.\ Cassella--Gulen 2018, bear vs.\ bull markets).
\end{itemize}

\vspace{0.6em}
\textbf{Salience channel.} Dispersed contest cross-sections, heavy news coverage $\Rightarrow$ stronger ($\lambda_1\uparrow$) and longer-lived ($\lambda_2\uparrow$) extrapolation.

\begin{frame}[shrink=26]{Extrapolative Beliefs: Biased Beliefs, Predictability, and Trading Profits}
\begin{columns}[T]
\column{0.52\textwidth}
\textbf{Beliefs negatively predict returns.}
\begin{itemize}
  \item Higher consensus score $\Rightarrow$ lower next-week return (Fama--MacBeth); holds for the extrapolation-predicted \emph{and} residual components --- beyond short-term reversal.
  \item Stronger among low-institutional-ownership and high-$\lambda_1(1-\lambda_2)$ stocks.
\end{itemize}

\vspace{0.5em}
\textbf{Model.} Extrapolators trade on beliefs; risk-averse fundamental traders only partially correct mispricing $\Rightarrow$ belief-driven overpricing reverses.

\vspace{0.5em}
\textbf{Strategy: long low-score, short high-score.}
\begin{itemize}
  \item $\sim$7 bps/day $\approx$ 18\%/yr, net of FF5 + momentum + short-term reversal.
  \item Out-of-sample: predicted scores beat 1-week and 1-month reversal sorts, even among the largest, most liquid non-Forcerank stocks.
\end{itemize}

\textbf{Takeaways}
\begin{enumerate}
  \item First direct cross-sectional evidence of \emph{return} (not cash-flow) extrapolation for individual stocks.
  \item Extrapolative beliefs are a distinct driver of short-term reversals, beyond liquidity shocks.
  \item Limitation: one-week horizon; silent on medium/long-run anomalies without further assumptions.
\end{enumerate}

% --- Slide 19: Conservatism ---
\begin{frame}{Psychology of Beliefs: Conservatism}
\begin{itemize}
\item Investors with extrapolative or overconfident beliefs adjust their views too much in response to new information and cause excessive movements in prices
\item But some phenomena --- momentum and post-earnings announcement drift (PEAD) --- may instead require that investors adjust their views \emph{too little}
\item \textbf{Conservatism}: individuals are too slow to update their beliefs in response to new information (Edwards, 1968)
\end{itemize}
\end{frame}

% --- Slide 20: Belief Perseverance ---
\begin{frame}{Psychology of Beliefs: Belief Perseverance and Confirmation Bias}
\begin{itemize}
\item \textbf{Belief perseverance:}
  \begin{itemize}
  \item Once we have formed an opinion, we are often too slow to change it on receipt of new evidence
  \item We do not look for evidence that would falsify our beliefs
  \item We ignore evidence that goes against our beliefs
  \end{itemize}
\item \textbf{Confirmation bias:}
  \begin{itemize}
  \item We actively seek out evidence that supports our beliefs
  \item We interpret ambiguous evidence as supportive of our beliefs
  \end{itemize}
\end{itemize}
\end{frame}

% --- Slide 21: Anchoring ---
\begin{frame}{Psychology of Beliefs: Anchoring}
\begin{itemize}
\item Kahneman and Tversky (1974): when people form numerical estimates of uncertain quantities, their adjustments away from an initial value (called the ``anchor'') are often insufficient
\begin{itemize}
\item E.g.\ estimates of the percentage of countries in the United Nations that are African countries were affected by a number between 0 and 100 that was determined by spinning a wheel of fortune in the subjects' presence
\item Those who were asked to compare their estimate to 10 subsequently estimated 25 percent, while those who compared it to 60 estimated 45 percent
\end{itemize} 
\item Applications in Finance:
\end{itemize}
\begin{itemize}
\item Anchoring on the 52-week high in the stock market
\item George and Hwang (2004): nearness to the 52-week high predicts future returns
\item They argue this is because traders are slow to react to good news when the price is close to the 52-week high
\item This also explains the momentum effect
\end{itemize} 
\end{frame}

\begin{frame}{The Effect of Feelings}
\begin{itemize}
\item A recent strand of research investigates how feelings affect beliefs and preferences
  \begin{itemize}
  \item Affective feelings (mood, emotion)
  \item Cognitive feelings
  \end{itemize}
\item \textbf{Affective feelings}
  \begin{itemize}
  \item A happier mood caused by an exogenous stimulus leads to more optimistic judgments about unrelated future outcomes (Johnson and Tversky, 1983) 
  \item E.g.\ sunshine, sports outcomes, air pollution
  \end{itemize}
\end{itemize}
\end{frame}

% --- Slide 23: Affective Feelings - Sports ---
\begin{frame}{Affective Feelings: Applications}
\begin{itemize}
\item \textbf{Sports}
\item Edmans, Garc\'{\i}a, and Norli (2007): when the national soccer team loses a World Cup match, the national stock market falls the next day
\item Disappointing sports news worsens the national mood, leading people to form more negative judgments about something unrelated, namely the economy's prospects
\item \textbf{Sunshine/Daylight}
\item Hirshleifer and Shumway (2003): the stock market has higher returns on sunnier days
\item Kamstra, Kramer, and Levi (2003): fewer hours of daylight lead to a type of depression, Seasonal Affective Disorder (SAD), which, in turn, leads to risk aversion and lower returns
\end{itemize}

\item These findings point to a more general mechanism that may amplify market fluctuations  
\begin{itemize}
\item If the stock market goes up, this improves the mood of many investors, leading them to form more positive judgments about the future prospects of the economy

\end{itemize}
\end{itemize}

  \item Local bias in stock ownership

% --- Slide 29: Sources ---
\begin{frame}{Sources of Heterogeneous Beliefs}
\begin{itemize}
\item What may cause heterogeneous beliefs?
\item Asymmetric information alone cannot cause rational agents with a common prior to agree to disagree
  \begin{itemize}
  \item e.g., the no-trade theorem of Milgrom and Stokey (1981)
  \end{itemize}
\item Heterogeneous beliefs must come from:
  \begin{itemize}
  \item Differences in prior beliefs: it is reasonable for agents to hold different priors about new things, e.g., Morris (1995) 
  \item Differences in how agents process information: investors put different weights on the same piece of information 
  \item The root cause can come from overconfidence, culture, or experience
  \end{itemize}
\end{itemize}
\end{frame}

% --- Slide 33: No SSC ---
\begin{frame}{Heterogeneous Beliefs (no SSC)}

\item Miller (1977): heterogeneous beliefs plus a short-sale constraint lead to overpricing
\end{itemize}

% --- Slide 35: Harrison and Kreps ---
\begin{frame}{Heterogeneous Beliefs with SSC, ctd.}
\begin{itemize}
\item \textbf{Dynamic argument}
\item Harrison and Kreps (1978)
\item If there are heterogeneous beliefs, you are willing to pay more than your own valuation of the asset today
  \begin{itemize}
  \item When information is released, there is a good chance that you will be able to sell to someone more optimistic 
  \item This creates a ``speculative'' component to the price
  \end{itemize}
 \item Scheinkman and Xiong (2003) build on this idea and develop a ``resale-option theory'' of speculative bubbles 
 \item When short-sales are constrained, an asset owner values the option to resell the asset to other, more optimistic buyers in the future
 
\end{itemize}
\end{frame}

\begin{frame}{Heterogeneous Beliefs with SSC, ctd.}
\begin{itemize}
\item \textbf{An example:} $t=0,1,2$
\item Risk-neutral investors A and B trade a risky asset
\item The interest rate is zero and there is no short selling
\end{itemize}
\vfill
\centering
\includegraphics[width=0.70\textwidth]{figures/resale_option.png}
\end{frame} 

% --- Slide 37: Example ctd ---
\begin{frame}{Heterogeneous Beliefs with SSC, ctd.}
\begin{itemize}
\item Both agents' buy-and-hold valuations of the asset are 50
\item What is the traded price?
  \begin{itemize}
  \item In state $u$, A sets the price at 90
  \item In state $d$, B sets the price at 25
  \item Then, at time 0, both value the asset at 57.5!
  \end{itemize}
\item This is because both expect to resell the asset to the other agent at a higher price in the future.
  \begin{itemize}
\item B expects to resell to A in state u at 90, which is higher than his valuation 75.
\item A expects to resell to B in state d at 25, which is higher than his valuation 10.
  \end{itemize} 
\item This resale option contributes to both A and B's initial asset valuation: a speculative bubble component.
\end{itemize}
\end{frame}

% --- Slide 38: Bubble Characteristics ---
\begin{frame}{Heterogeneous Beliefs with SSC, ctd.}
\begin{itemize}
\item Characteristics of this bubble component:
\item The valuation of the resale option is determined by fluctuations in agents' beliefs and asset float
\item Scheinkman and Xiong (2003): the more likely their beliefs flip, the more they trade and the more valuable the resale option. 
\item This is why bubbles tend to coincide with trading frenzies. 
\item Another key factor is asset float: more shares available for trading $\Rightarrow$ smaller bubble
\begin{itemize}
\item This is because when investors have downward-sloping demand curves, an owner can sell his share only if the buyer's belief passes his by a sufficient margin 
\item A larger float makes this less likely to happen 
\end{itemize} 
\end{itemize}
\end{frame}

\item \textbf{The coincidence of high valuations and heavy trading}
\item Hong and Stein (2007): value and growth stocks diverge in both valuation and turnover during the tech bubble period

% --- Slide 42a: Machine Forecast Disagreement (Bali, Kelly, M\"orke, Rahman, RFS 2026) ---
\begin{frame}[shrink=26]{Machine Forecast Disagreement}
\footnotesize
\setlength{\parskip}{6pt}
\setlength{\leftmargini}{1.1em}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{\color{CityUBlue}Motivation: beliefs are unobservable}
\begin{itemize}
    \setlength{\itemsep}{4pt}
  \item Disagreement drives trading and mispricing: short-sale constraints shut pessimists out $\Rightarrow$ high-disagreement stocks are overpriced (Miller 1977).
  \item Classic proxy AFD (analyst dispersion, DMS 2002): limited I/B/E/S coverage; behavioral biases (Johnson 2004).
\end{itemize}

\textbf{\color{CityUBlue}Construction: MFD}
\begin{itemize}
    \setlength{\itemsep}{4pt}
  \item Each investor $k$ is a random forest: $E_{k,t}[r_{i,t+1}] = RF_k(z_{k,i,t})$.
  \item $K{=}100$ investors share one info set (152 chars.) but each draws a random $50\%$ feature subset (info.\ differences) + bootstrap/dropout (prior differences) $\Rightarrow$ spans all three disagreement sources (Barberis 2018).
  \item Investors are sophisticated yet imperfect; $\mathrm{MFD}_{i,t}=$ dispersion of forecasts across investors.
  \item 10-yr rolling window, refit annually; full-sample; beats AFD in predictive power.
\end{itemize}

\column{0.42\textwidth}
\textbf{\color{CityUBlue}Results}
\begin{itemize}
    \setlength{\itemsep}{4pt}
  \item Decile long-short (buy low MFD) earns \textbf{\color{CityURed}13\%} p.a.\ value-weighted.
  \item Earnings-based MFD also predicts; correlates more with AFD.
\end{itemize}

\textbf{\color{CityUBlue}Mechanism: short-sale constraints, limits to arbitrage}
\begin{itemize}
    \setlength{\itemsep}{4pt}
  \item High borrow-fee $-2.36\%$/mo vs.\ low-fee $-0.12\%$ (diff.\ $-2.24$, $t=-6.06$).
  \item High retail ownership $-1.10\%$ vs.\ inst.\ $-0.31\%$ ($t=4.53$).
  \item High-MISP group (Stambaugh--Yu--Yuan): $-0.68\%$ vs.\ $-0.31\%$.
  \item Correction concentrated on earnings days: long-short $156\%/113\%$ higher in 1/3-day windows $\Rightarrow$ arbitrage limits amplify mispricing.
\end{itemize}

\par\noindent\textbf{\color{CityUBlue}Takeaway:} disagreement can be \textbf{simulated, not surveyed} --- heterogeneous yet rational models generate a belief distribution; high-MFD stocks are overpriced under short-sale constraints $\Rightarrow$ the discount is mispricing, not risk compensation.

\vspace{6pt}
{\footnotesize Bali, Kelly, M\"orke \& Rahman, \emph{Machine Forecast Disagreement}, RFS 2026 (robust to 19 parameterizations).}
\end{columns}

		\boldsymbol{d}_i = \boldsymbol{\beta}_i(\boldsymbol{Z}) + \boldsymbol{e}_i
		\]

\begin{frame}{The Causal Effect of Disagreement on Asset Prices}

\begin{frame}{Biased Beliefs: Concerns}
\begin{itemize}
\item Do people simply learn their way out of biased beliefs?
\item Only to a limited extent
\item Many of the most persistent behavioral biases are mental shortcuts that were useful for survival over human evolution
  \begin{itemize}
  \item ``Loss aversion'' increases survival probability
  \item Overconfidence increases effort and trust within your community  
  \end{itemize}
\item Do highly trained professionals exhibit these biases? 
  \begin{itemize}
  \item Yes, professionals sometimes do worse
  \item CEOs and founders are more overconfident than the average person  
  \end{itemize}
 \item Do people exhibit these biases even when stakes are high?
 
\end{itemize}
